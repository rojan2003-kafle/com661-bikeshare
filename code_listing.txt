import math
import re
from datetime import datetime, timezone

from bson import ObjectId
from flask import request, make_response, jsonify


class ApiError(Exception):
    """raise ApiError(400, "message") anywhere -> app.py turns it into a JSON error"""
    def __init__(self, status, message):
        super().__init__(message)
        self.status = status
        self.message = message


def now():
    return datetime.now(timezone.utc)


def to_json(value):
    """MongoDB gives us ObjectId and datetime objects - jsonify can't handle those,
    so convert them to strings (and never send password hashes back)"""
    if isinstance(value, dict):
        return {k: to_json(v) for k, v in value.items() if k != "password_hash"}
    if isinstance(value, list):
        return [to_json(v) for v in value]
    if isinstance(value, ObjectId):
        return str(value)
    if isinstance(value, datetime):
        return value.astimezone(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    return value


def respond(body, status=200):
    return make_response(jsonify(to_json(body)), status)


# ---------------- validators: check type AND range ----------------
def parse_int(value, name, min_val=None, max_val=None):
    try:
        number = int(str(value).strip())
    except ValueError:
        raise ApiError(400, f"'{name}' must be a whole number")
    if min_val is not None and number < min_val:
        raise ApiError(400, f"'{name}' must be at least {min_val}")
    if max_val is not None and number > max_val:
        raise ApiError(400, f"'{name}' must be no more than {max_val}")
    return number


def parse_enum(value, name, allowed):
    value = str(value).strip().lower()
    if value not in allowed:
        raise ApiError(400, f"'{name}' must be one of: {', '.join(allowed)}")
    return value


def check_id(value, prefix, name):
    """our IDs look like ST001, BK0001, U001 ..."""
    if not re.fullmatch(prefix + r"\d{3,6}", value):
        raise ApiError(400, f"Invalid {name} ID format '{value}' (expected e.g. {prefix}001)")
    return value


# ---------------- pagination ----------------
def get_pagination(default_size=10, max_size=100):
    page_num = parse_int(request.args.get("pn", 1), "pn", 1)
    page_size = parse_int(request.args.get("ps", default_size), "ps", 1, max_size)
    return page_num, page_size, (page_num - 1) * page_size


def paged_response(items, total, page_num, page_size):
    return {"page": page_num, "page_size": page_size, "total": total,
            "total_pages": math.ceil(total / page_size), "data": items}

# ---------------- reading the request body ----------------
def get_data():
    """accept form-data (like the module) OR a JSON body"""
    data = request.get_json(silent=True)
    if isinstance(data, dict):
        return data
    return request.form.to_dict()


def require_fields(data, fields):
    missing = [f for f in fields if f not in data or str(data[f]).strip() == ""]
    if missing:
        raise ApiError(400, "Missing required field(s): " + ", ".join(missing))


def parse_str(value, name, min_len=1, max_len=200):
    value = str(value).strip()
    if not (min_len <= len(value) <= max_len):
        raise ApiError(400, f"'{name}' must be between {min_len} and {max_len} characters")
    return value


def parse_email(value):
    value = parse_str(value, "email", 5, 120).lower()
    if not re.match(r"^[^@\s]+@[^@\s]+\.[a-zA-Z]{2,}$", value):
        raise ApiError(400, "'email' is not a valid email address")
    return value


def parse_password(value):
    value = parse_str(value, "password", 8, 64)
    if not (re.search(r"[A-Za-z]", value) and re.search(r"\d", value)):
        raise ApiError(400, "'password' must contain at least one letter and one number")
    return value


# ---------------- readable IDs: U151, ST121, BK0901 ----------------
def next_id(counter_name, collection, field, prefix, width):
    import globals
    from pymongo import ReturnDocument
    if globals.counters.find_one({"_id": counter_name}) is None:
        # first time: start counting from the highest ID already in the collection
        last = collection.find_one({field: {"$regex": "^" + prefix}}, sort=[(field, -1)])
        start = int(last[field][len(prefix):]) if last else 0
        globals.counters.insert_one({"_id": counter_name, "seq": start})
    # $inc is atomic: two requests at the same time never get the same number
    doc = globals.counters.find_one_and_update(
        {"_id": counter_name}, {"$inc": {"seq": 1}}, return_document=ReturnDocument.AFTER)
    return f"{prefix}{doc['seq']:0{width}d}"


def parse_float(value, name, min_val=None, max_val=None):
    try:
        number = float(str(value).strip())
    except ValueError:
        raise ApiError(400, f"'{name}' must be a number")
    if min_val is not None and number < min_val:
        raise ApiError(400, f"'{name}' must be at least {min_val}")
    if max_val is not None and number > max_val:
        raise ApiError(400, f"'{name}' must be no more than {max_val}")
    return number

===== decorators.py =====
from functools import wraps

import jwt
from flask import request, g

import globals
from utils import ApiError


def jwt_required(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        token = request.headers.get("x-access-token")
        if not token:
            raise ApiError(401, "Token is missing - log in and send it in the x-access-token header")
        try:
            data = jwt.decode(token, globals.secret_key, algorithms=["HS256"])
        except jwt.ExpiredSignatureError:
            raise ApiError(401, "Token has expired - please log in again")
        except jwt.InvalidTokenError:
            raise ApiError(401, "Token is invalid")
        if globals.blacklist.find_one({"token": token}):
            raise ApiError(401, "Token has been cancelled (logged out)")
        # remember who is logged in for the rest of this request
        g.user_id = data["user_id"]
        g.role = data["role"]
        g.token = token
        return func(*args, **kwargs)
    return wrapper


def role_required(*roles):
    """@role_required("admin")  or  @role_required("mechanic", "admin")"""
    def decorator(func):
        @wraps(func)
        @jwt_required
        def wrapper(*args, **kwargs):
            if g.role not in roles:
                raise ApiError(403, f"Access denied - this action needs role: {' or '.join(roles)}")
            return func(*args, **kwargs)
        return wrapper
    return decorator

from flask import Blueprint, request

admin_bp = Blueprint("admin_bp", __name__)

WEEKDAYS = {1: "Sunday", 2: "Monday", 3: "Tuesday", 4: "Wednesday",
            5: "Thursday", 6: "Friday", 7: "Saturday"}


def since():
    """?days=30 -> only look at the last 30 days"""
    days = parse_int(request.args.get("days", 90), "days", 1, 3650)
    return now() - timedelta(days=days), days


def station_names():
    return {s["station_id"]: s["name"] for s in globals.stations.find({}, {"station_id": 1, "name": 1})}


# ---------------- REVENUE ----------------
@admin_bp.route("/admin/revenue", methods=["GET"])
@role_required("admin")
def revenue():
    """/admin/revenue?group_by=day|month&days=90"""
    group_by = parse_enum(request.args.get("group_by", "month"), "group_by", ["day", "month"])
    fmt = "%Y-%m-%d" if group_by == "day" else "%Y-%m"
    start, days = since()

    result = list(globals.rentals.aggregate([
        {"$match": {"payment.status": "succeeded", "payment.paid_at": {"$gte": start}}},
        # join the bike (for its type) and the start station (for its zone)
        {"$lookup": {"from": "bikes", "localField": "bike_id", "foreignField": "bike_id", "as": "bike"}},
        {"$lookup": {"from": "stations", "localField": "start_station_id",
                     "foreignField": "station_id", "as": "station"}},
        {"$addFields": {"bike_type": {"$arrayElemAt": ["$bike.type", 0]},
                        "zone": {"$arrayElemAt": ["$station.zone", 0]}}},
        # $facet = several reports from ONE query
        {"$facet": {
            "totals": [{"$group": {"_id": None, "revenue": {"$sum": "$payment.amount"},
                                   "payments": {"$sum": 1}, "avg": {"$avg": "$payment.amount"}}}],
            "by_period": [{"$group": {"_id": {"$dateToString": {"format": fmt, "date": "$payment.paid_at"}},
                                      "revenue": {"$sum": "$payment.amount"}, "rides": {"$sum": 1}}},
                          {"$sort": {"_id": 1}}],
            "by_bike_type": [{"$group": {"_id": "$bike_type", "revenue": {"$sum": "$payment.amount"},
                                         "rides": {"$sum": 1}}}, {"$sort": {"revenue": -1}}],
            "by_zone": [{"$group": {"_id": "$zone", "revenue": {"$sum": "$payment.amount"},
                                    "rides": {"$sum": 1}}}, {"$sort": {"revenue": -1}}],
        }},
    ]))[0]

    t = result["totals"][0] if result["totals"] else {"revenue": 0, "payments": 0, "avg": 0}
    rows = lambda items, key: [{key: x["_id"], "revenue": round(x["revenue"], 2), "rides": x["rides"]}
                               for x in items]
    return respond({"days": days, "currency": "GBP",
                    "total_revenue": round(t["revenue"], 2), "payments": t["payments"],
                    "average_payment": round(t["avg"] or 0, 2),
                    "by_" + group_by: rows(result["by_period"], group_by),
                    "by_bike_type": rows(result["by_bike_type"], "bike_type"),
                    "by_zone": rows(result["by_zone"], "zone")}, 200)

@admin_bp.route("/admin/analytics", methods=["GET"])
@role_required("admin")
def analytics():
    start, days = since()
    result = list(globals.rentals.aggregate([
        {"$match": {"status": "completed", "start_time": {"$gte": start}}},
        {"$facet": {
            "hours": [{"$group": {"_id": {"$hour": "$start_time"}, "rides": {"$sum": 1}}},
                      {"$sort": {"_id": 1}}],
            "weekdays": [{"$group": {"_id": {"$dayOfWeek": "$start_time"}, "rides": {"$sum": 1}}},
                         {"$sort": {"_id": 1}}],
            "routes": [
                {"$match": {"$expr": {"$ne": ["$start_station_id", "$end_station_id"]}}},
                {"$group": {"_id": {"from": "$start_station_id", "to": "$end_station_id"},
                            "rides": {"$sum": 1}, "avg_minutes": {"$avg": "$duration_min"}}},
                {"$sort": {"rides": -1}}, {"$limit": 10}],
        }},
    ]))[0]
    names = station_names()
    hours = [{"hour": f"{x['_id']:02d}:00", "rides": x["rides"]} for x in result["hours"]]
    return respond({
        "days": days,
        "busiest_hour": max(hours, key=lambda h: h["rides"]) if hours else None,
        "rides_by_hour": hours,
        "rides_by_weekday": [{"day": WEEKDAYS[x["_id"]], "rides": x["rides"]} for x in result["weekdays"]],
        "popular_routes": [{"from": names.get(x["_id"]["from"]), "to": names.get(x["_id"]["to"]),
                            "rides": x["rides"], "avg_minutes": round(x["avg_minutes"], 1)}
                           for x in result["routes"]],
    }, 200)


# ---------------- REBALANCING ----------------
@admin_bp.route("/admin/rebalancing", methods=["GET"])
@role_required("admin")
def rebalancing():
    """
    Stations under ?low=20% full need bikes, stations over ?high=85% have too many.
    Pair each overfull station with the NEAREST under-full one and suggest how many bikes to move.
    """
    low = parse_int(request.args.get("low", 20), "low", 0, 50)
    high = parse_int(request.args.get("high", 85), "high", 50, 100)
    stations = list(globals.stations.aggregate([{"$match": {}}] + availability_stages()))
    for s in stations:
        s["occupancy_pct"] = round(100 * s["bikes_docked"] / s["total_docks"], 1)
    empty = [s for s in stations if s["occupancy_pct"] <= low]
    full = sorted([s for s in stations if s["occupancy_pct"] >= high], key=lambda s: -s["occupancy_pct"])

    spare = {s["station_id"]: s["bikes_available"] - s["total_docks"] // 2 for s in full}
    need = {s["station_id"]: s["total_docks"] // 2 - s["bikes_docked"] for s in empty}
    moves = []
    for src in full:
        lng1, lat1 = src["location"]["coordinates"]
        for dst in sorted(empty, key=lambda d: haversine_km(lat1, lng1, d["location"]["coordinates"][1],
                                                            d["location"]["coordinates"][0])):
            qty = min(spare[src["station_id"]], need[dst["station_id"]])
            if qty <= 0:
                continue
            lng2, lat2 = dst["location"]["coordinates"]
            moves.append({"from": src["name"], "to": dst["name"], "bikes": qty,
                          "distance_km": round(haversine_km(lat1, lng1, lat2, lng2), 2)})
            spare[src["station_id"]] -= qty
            need[dst["station_id"]] -= qty
            if spare[src["station_id"]] <= 0:
                break

    short = lambda s: {k: s[k] for k in ["station_id", "name", "total_docks", "bikes_docked",
                                          "free_docks", "occupancy_pct"]}
    return respond({"nearly_empty": [short(s) for s in empty],
                    "nearly_full": [short(s) for s in full],
                    "suggested_moves": moves,
                    "bikes_to_move": sum(m["bikes"] for m in moves)}, 200)


# ---------------- REPEAT FAULTS ----------------
@admin_bp.route("/admin/repeat-faults", methods=["GET"])
@role_required("admin", "mechanic")
def repeat_faults():
    """bikes with ?min_reports=3 or more faults in the last ?days=90"""
    min_reports = parse_int(request.args.get("min_reports", 3), "min_reports", 2, 50)
    start, days = since()
    result = list(globals.maintenance.aggregate([
        {"$match": {"created_at": {"$gte": start}}},
        {"$group": {"_id": "$bike_id", "reports": {"$sum": 1},
                    "issues": {"$push": "$issue_type"}, "last_reported": {"$max": "$created_at"}}},
        {"$match": {"reports": {"$gte": min_reports}}},
        {"$sort": {"reports": -1}},
        {"$project": {"_id": 0, "bike_id": "$_id", "reports": 1, "issues": 1, "last_reported": 1}},
    ]))
    for r in result:
        r["recommendation"] = "consider retiring" if r["reports"] >= 5 else "full inspection"
    return respond({"days": days, "count": len(result), "bikes": result}, 200)

from datetime import timedelta

import bcrypt
import jwt
from flask import Blueprint, request, g

import globals
from decorators import jwt_required
from utils import (ApiError, respond, get_data, require_fields, parse_str,
                   parse_email, parse_password, next_id, now)

auth_bp = Blueprint("auth_bp", __name__)


@auth_bp.route("/auth/register", methods=["POST"])
def register():
    data = get_data()
    require_fields(data, ["name", "email", "password"])
    name = parse_str(data["name"], "name", 2, 60)
    email = parse_email(data["email"])
    password = parse_password(data["password"])

    if globals.users.find_one({"email": email}):
        raise ApiError(409, "An account with this email already exists")

    new_user = {
        "user_id": next_id("users", globals.users, "user_id", "U", 3),
        "name": name,
        "email": email,
        "password_hash": bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode(),
        "role": "rider",            # everyone signs up as a rider; admins promote staff
        "created_at": now(),
    }
    globals.users.insert_one(new_user)
    new_user.pop("_id", None)
    return respond({"message": "Account created", "user": new_user}, 201)


@auth_bp.route("/auth/login", methods=["POST"])
def login():
    auth = request.authorization       # Basic Auth: username = email
    if not auth or not auth.username or not auth.password:
        raise ApiError(401, "Login required - send email and password using Basic Auth")

    user = globals.users.find_one({"email": auth.username.strip().lower()})
    if user is None or not bcrypt.checkpw(auth.password.encode(), user["password_hash"].encode()):
        raise ApiError(401, "Incorrect email or password")

    token = jwt.encode({"user_id": user["user_id"], "role": user["role"],
                        "exp": now() + timedelta(minutes=60)},
                       globals.secret_key, algorithm="HS256")
    return respond({"token": token, "user_id": user["user_id"], "name": user["name"],
                    "role": user["role"]}, 200)


@auth_bp.route("/auth/logout", methods=["POST"])
@jwt_required
def logout():
    globals.blacklist.insert_one({"token": g.token, "user_id": g.user_id, "logged_out_at": now()})
    return respond({"message": "Logged out"}, 200)


@auth_bp.route("/auth/me", methods=["GET"])
@jwt_required
def my_profile():
    user = globals.users.find_one({"user_id": g.user_id}, {"_id": 0, "password_hash": 0})
    if user is None:
        raise ApiError(404, "User no longer exists")
    return respond(user, 200)

from flask import Blueprint, request
import globals
from decorators import jwt_required, role_required
from utils import (ApiError, respond, get_data, require_fields, parse_int, parse_enum,
                   check_id, get_pagination, paged_response, next_id)

bikes_bp = Blueprint("bikes_bp", __name__)

TYPES = ["standard", "electric"]
STATUSES = ["available", "in_use", "reserved", "under_repair"]


def find_bike(bike_id):
    check_id(bike_id, "BK", "bike")
    bike = globals.bikes.find_one({"bike_id": bike_id}, {"_id": 0})
    if bike is None:
        raise ApiError(404, f"Bike {bike_id} not found")
    return bike


def check_free_dock(station_id):
    """a bike can only be placed at a station that exists and has an empty dock"""
    check_id(station_id, "ST", "station")
    station = globals.stations.find_one({"station_id": station_id})
    if station is None:
        raise ApiError(404, f"Station {station_id} not found")
    docked = globals.bikes.count_documents({"station_id": station_id})
    if docked >= station["total_docks"]:
        raise ApiError(409, f"Station {station_id} is full ({docked}/{station['total_docks']} docks used)")


# ---------- GET all bikes with filters: ?status=&type=&station_id=&min_battery= ----------
@bikes_bp.route("/bikes", methods=["GET"])
@jwt_required
def show_all_bikes():
    page_num, page_size, page_start = get_pagination()
    query = {}
    if request.args.get("status"):
        query["status"] = parse_enum(request.args["status"], "status", STATUSES)
    if request.args.get("type"):
        query["type"] = parse_enum(request.args["type"], "type", TYPES)
    if request.args.get("station_id"):
        query["station_id"] = check_id(request.args["station_id"], "ST", "station")
    if request.args.get("min_battery"):
        query["battery_level"] = {"$gte": parse_int(request.args["min_battery"], "min_battery", 0, 100)}

    total = globals.bikes.count_documents(query)
    data = list(globals.bikes.find(query, {"_id": 0}).sort("bike_id", 1).skip(page_start).limit(page_size))
    return respond(paged_response(data, total, page_num, page_size), 200)

@jwt_required
def show_one_bike(bike_id):
    return respond(find_bike(bike_id), 200)


# ---------- ADMIN: add / edit / delete ----------
@bikes_bp.route("/bikes", methods=["POST"])
@role_required("admin")
def add_bike():
    data = get_data()
    require_fields(data, ["type", "station_id"])
    bike_type = parse_enum(data["type"], "type", TYPES)
    station_id = str(data["station_id"]).strip()
    check_free_dock(station_id)

    battery = None
    if bike_type == "electric":
        battery = parse_int(data.get("battery_level", 100), "battery_level", 0, 100)
    elif data.get("battery_level"):
        raise ApiError(400, "'battery_level' only applies to electric bikes")

    new_bike = {"bike_id": next_id("bikes", globals.bikes, "bike_id", "BK", 4),
                "station_id": station_id, "type": bike_type,
                "battery_level": battery, "status": "available"}
    globals.bikes.insert_one(new_bike)
    new_bike.pop("_id", None)
    return respond({"message": "Bike added",
                    "url": request.host_url + "api/v1.0/bikes/" + new_bike["bike_id"],
                    "bike": new_bike}, 201)


@bikes_bp.route("/bikes/<string:bike_id>", methods=["PUT"])
@role_required("admin")
def edit_bike(bike_id):
    """admin can move a bike (rebalancing), change its battery or take it out of service"""
    bike = find_bike(bike_id)
    if bike["status"] in ("in_use", "reserved"):
        raise ApiError(409, f"Bike {bike_id} is {bike['status']} - wait until the ride ends")
    data = get_data()
    changes = {}
    if "battery_level" in data:
        if bike["type"] != "electric":
            raise ApiError(400, "'battery_level' only applies to electric bikes")
        changes["battery_level"] = parse_int(data["battery_level"], "battery_level", 0, 100)
    if "status" in data:
        changes["status"] = parse_enum(data["status"], "status", ["available", "under_repair"])
    if "station_id" in data:
        new_station = str(data["station_id"]).strip()
        if new_station != bike["station_id"]:
            check_free_dock(new_station)
        changes["station_id"] = new_station
    if not changes:
        raise ApiError(400, "Nothing to update - send battery_level, status and/or station_id")
    globals.bikes.update_one({"bike_id": bike_id}, {"$set": changes})
    return respond({"message": "Bike updated", "bike": find_bike(bike_id)}, 200)


@bikes_bp.route("/bikes/<string:bike_id>", methods=["DELETE"])

import math
from datetime import timedelta

from flask import Blueprint, request, g
from pymongo import ReturnDocument

import globals
from decorators import jwt_required, role_required
from utils import (ApiError, respond, get_data, require_fields, check_id,
                   get_pagination, paged_response, next_id, now)
from blueprints.bikes.bikes import find_bike

rentals_bp = Blueprint("rentals_bp", __name__)

# ---------- pricing ----------
UNLOCK_FEE = 1.00
PER_MIN = {"standard": 0.05, "electric": 0.20}
OUTER_ZONE_DISCOUNT = 0.10        # 10% off rides starting in the outer zone
RESERVATION_MINUTES = 10
MIN_BATTERY = 20


def calculate_cost(bike_type, minutes, zone):
    cost = UNLOCK_FEE + PER_MIN[bike_type] * minutes
    if zone == "outer":
        cost *= (1 - OUTER_ZONE_DISCOUNT)
    return round(cost, 2)


def haversine_km(lat1, lng1, lat2, lng2):
    """straight-line distance between two GPS points on the Earth"""
    r = 6371
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dp, dl = math.radians(lat2 - lat1), math.radians(lng2 - lng1)
    a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    return 2 * r * math.asin(math.sqrt(a))


# ---------- helpers ----------
def expire_old_reservations():
    """reservations past their 10 minutes are cancelled and the bike freed (lazy expiry)"""
    for r in globals.rentals.find({"status": "reserved", "expires_at": {"$lt": now()}}):
        globals.rentals.update_one({"rental_id": r["rental_id"]},
                                   {"$set": {"status": "cancelled", "cancel_reason": "expired"}})
        globals.bikes.update_one({"bike_id": r["bike_id"], "status": "reserved"},
                                 {"$set": {"status": "available"}})


def check_no_open_rental(user_id):
    current = globals.rentals.find_one({"user_id": user_id, "status": {"$in": ["reserved", "active"]}})
    if current:
        raise ApiError(409, f"You already have a {current['status']} rental ({current['rental_id']}) "
                            "- finish or cancel it first")


def grab_bike(bike_id, new_status):
    """
    ATOMIC: only changes the bike if it is STILL 'available' at the moment of the update,
    so two riders pressing 'reserve' at the same time can never both get the same bike.
    """
    bike = find_bike(bike_id)
    if bike["type"] == "electric" and (bike["battery_level"] or 0) < MIN_BATTERY:
        raise ApiError(409, f"Bike {bike_id} battery is too low ({bike['battery_level']}%)")
    updated = globals.bikes.find_one_and_update(
        {"bike_id": bike_id, "status": "available"},
        {"$set": {"status": new_status}}, return_document=ReturnDocument.AFTER)
    if updated is None:
        raise ApiError(409, f"Bike {bike_id} is not available (it is {bike['status']})")
    return updated


def find_rental(rental_id):
    check_id(rental_id, "R", "rental")
    rental = globals.rentals.find_one({"rental_id": rental_id}, {"_id": 0})
    if rental is None:
        raise ApiError(404, f"Rental {rental_id} not found")
    if rental["user_id"] != g.user_id and g.role != "admin":
        raise ApiError(403, "This rental belongs to another user")
    return rental


def new_rental(bike, status):
    t = now()
    rental = {
        "rental_id": next_id("rentals", globals.rentals, "rental_id", "R", 5),
        "user_id": g.user_id, "bike_id": bike["bike_id"],
        "start_station_id": bike["station_id"], "end_station_id": None,
        "reserved_at": t if status == "reserved" else None,
        "expires_at": t + timedelta(minutes=RESERVATION_MINUTES) if status == "reserved" else None,
        "start_time": t if status == "active" else None, "end_time": None,
        "distance_km": 0, "duration_min": 0, "total_cost": 0,
        "status": status, "payment": None,
    }
    globals.rentals.insert_one(rental)
    rental.pop("_id", None)
    return rental

@rentals_bp.route("/rentals/<string:rental_id>/return", methods=["POST"])
@role_required("rider")
def return_bike(rental_id):
    data = get_data()
    require_fields(data, ["end_station_id"])
    end_id = check_id(str(data["end_station_id"]).strip(), "ST", "station")
    rental = find_rental(rental_id)
    if rental["status"] != "active":
        raise ApiError(409, f"Only an active rental can be returned - this one is {rental['status']}")
    end_station = globals.stations.find_one({"station_id": end_id})
    if end_station is None:
        raise ApiError(404, f"Station {end_id} not found")

    # station full? refuse and suggest the 3 closest stations with a free dock
    docked = globals.bikes.count_documents({"station_id": end_id})
    if docked >= end_station["total_docks"]:
        lng, lat = end_station["location"]["coordinates"]
        nearby = list(globals.stations.aggregate([
            {"$geoNear": {"near": {"type": "Point", "coordinates": [lng, lat]},
                          "distanceField": "distance_m", "maxDistance": 3000, "spherical": True}},
            {"$lookup": {"from": "bikes", "localField": "station_id", "foreignField": "station_id", "as": "d"}},
            {"$project": {"_id": 0, "station_id": 1, "name": 1, "distance_m": 1,
                          "free_docks": {"$subtract": ["$total_docks", {"$size": "$d"}]}}},
            {"$match": {"free_docks": {"$gt": 0}}},
            {"$limit": 3}]))
        for s in nearby:
            s["distance_m"] = round(s["distance_m"])
        return respond({"error": f"Station {end_id} is full - please return to another station",
                        "status": 409, "suggested_stations": nearby}, 409)

    bike = find_bike(rental["bike_id"])
    start_station = globals.stations.find_one({"station_id": rental["start_station_id"]})
    end_time = now()
    minutes = max(1, math.ceil((end_time - rental["start_time"]).total_seconds() / 60))
    a = start_station["location"]["coordinates"]
    b = end_station["location"]["coordinates"]
    distance = round(haversine_km(a[1], a[0], b[1], b[0]), 2)
    cost = calculate_cost(bike["type"], minutes, start_station["zone"])

    globals.rentals.update_one({"rental_id": rental_id}, {"$set": {
        "status": "completed", "end_station_id": end_id, "end_time": end_time,
        "duration_min": minutes, "distance_km": distance, "total_cost": cost,
        "payment": {"status": "unpaid", "amount": cost, "currency": "GBP"}}})

    bike_changes = {"station_id": end_id, "status": "available"}
    if bike["type"] == "electric":
        bike_changes["battery_level"] = max(0, bike["battery_level"] - math.ceil(distance * 3))
    globals.bikes.update_one({"bike_id": bike["bike_id"]}, {"$set": bike_changes})

    return respond({"message": f"Ride complete - {minutes} min, {distance} km, £{cost:.2f}",
                    "rental": find_rental(rental_id)}, 200)

    page_num, page_size, page_start = get_pagination()
    result = list(globals.rentals.aggregate([
        {"$match": {"user_id": g.user_id}},
        {"$facet": {
            "data": [
                {"$sort": {"rental_id": -1}}, {"$skip": page_start}, {"$limit": page_size},
                {"$lookup": {"from": "stations", "localField": "start_station_id",

import re
from flask import Blueprint, request
import globals
from utils import (ApiError, respond, parse_int, parse_float, parse_enum, check_id,
                   get_pagination, paged_response)

stations_bp = Blueprint("stations_bp", __name__)

ZONES = ["central", "inner", "outer"]
MIN_BATTERY = 20     # electric bikes below 20% can't be hired


def availability_stages():
    """
    Aggregation stages that JOIN each station to the bikes docked there ($lookup)
    and work out live numbers. A bike counts as available if its status is
    'available' and (for electric bikes) the battery is at least MIN_BATTERY.
    """
    usable = {"$and": [
        {"$eq": ["$$b.status", "available"]},
        {"$or": [{"$eq": ["$$b.type", "standard"]},
                 {"$gte": ["$$b.battery_level", MIN_BATTERY]}]},
    ]}
    return [
        {"$lookup": {"from": "bikes", "localField": "station_id",
                     "foreignField": "station_id", "as": "docked"}},
        {"$addFields": {
            "bikes_docked": {"$size": "$docked"},
            "bikes_available": {"$size": {"$filter": {"input": "$docked", "as": "b", "cond": usable}}},
        }},
        {"$addFields": {"free_docks": {"$subtract": ["$total_docks", "$bikes_docked"]}}},
        {"$project": {"_id": 0, "docked": 0}},
    ]

@stations_bp.route("/stations", methods=["GET"])
def show_all_stations():
    page_num, page_size, page_start = get_pagination()

    match = {}
    if request.args.get("zone"):
        match["zone"] = parse_enum(request.args["zone"], "zone", ZONES)
    if request.args.get("search"):
        text = re.escape(request.args["search"].strip())
        match["$or"] = [{"name": {"$regex": text, "$options": "i"}},
                        {"address": {"$regex": text, "$options": "i"}}]

    pipeline = [{"$match": match}] + availability_stages()
    if request.args.get("min_bikes"):
        pipeline.append({"$match": {"bikes_available":
                         {"$gte": parse_int(request.args["min_bikes"], "min_bikes", 0, 100)}}})

    # $facet = run two pipelines at once: one page of data + the total count
    pipeline.append({"$facet": {
        "data": [{"$sort": {"name": 1}}, {"$skip": page_start}, {"$limit": page_size}],
        "total": [{"$count": "count"}],
    }})
    result = list(globals.stations.aggregate(pipeline))[0]
    total = result["total"][0]["count"] if result["total"] else 0
    return respond(paged_response(result["data"], total, page_num, page_size), 200)

@stations_bp.route("/stations/nearby", methods=["GET"])
def nearby_stations():
    """ /stations/nearby?lat=51.5033&lng=-0.1145&radius=800&min_bikes=1 """
    if not request.args.get("lat") or not request.args.get("lng"):
        raise ApiError(400, "Query parameters 'lat' and 'lng' are required")
    lat = parse_float(request.args["lat"], "lat", -90, 90)
    lng = parse_float(request.args["lng"], "lng", -180, 180)
    radius = parse_int(request.args.get("radius", 1000), "radius", 50, 20000)   # metres
    min_bikes = parse_int(request.args.get("min_bikes", 0), "min_bikes", 0, 100)

    pipeline = [
        # $geoNear MUST be the first stage and needs the 2dsphere index
        {"$geoNear": {
            "near": {"type": "Point", "coordinates": [lng, lat]},   # GeoJSON = [lng, lat]
            "distanceField": "distance_m",
            "maxDistance": radius,
            "spherical": True,
        }},
    ] + availability_stages() + [
        {"$match": {"bikes_available": {"$gte": min_bikes}}},
        {"$limit": 10},
    ]
    results = list(globals.stations.aggregate(pipeline))
    for s in results:
        s["distance_m"] = round(s["distance_m"])
        s["walking_minutes"] = max(1, round(s["distance_m"] / 80))   # ~80 m per minute
    return respond({"count": len(results), "stations": results}, 200)

def read_station_fields(data, partial=False):
    """validate the body; partial=True for PUT (only send what you change)"""
    if not partial:
        require_fields(data, ["name", "address", "lat", "lng", "total_docks", "zone"])
    fields = {}
    if "name" in data:
        fields["name"] = parse_str(data["name"], "name", 3, 100)
    if "address" in data:
        fields["address"] = parse_str(data["address"], "address", 5, 200)
    if "lat" in data or "lng" in data:
        lat = parse_float(data.get("lat"), "lat", 51.25, 51.75)    # Greater London only
        lng = parse_float(data.get("lng"), "lng", -0.55, 0.35)
        fields["location"] = {"type": "Point", "coordinates": [lng, lat]}
    if "total_docks" in data:
        fields["total_docks"] = parse_int(data["total_docks"], "total_docks", 1, 100)
    if "zone" in data:
        fields["zone"] = parse_enum(data["zone"], "zone", ZONES)
    return fields

@stations_bp.route("/stations/<string:station_id>", methods=["PUT"])
@role_required("admin")
def edit_station(station_id):
    check_id(station_id, "ST", "station")
    fields = read_station_fields(get_data(), partial=True)
    if not fields:
        raise ApiError(400, "Nothing to update - send at least one field")
    if "total_docks" in fields:
        docked = globals.bikes.count_documents({"station_id": station_id})
        if fields["total_docks"] < docked:
            raise ApiError(409, f"{docked} bikes are docked here - total_docks can't be lower than that")
    result = globals.stations.update_one({"station_id": station_id}, {"$set": fields})
    if result.matched_count == 0:
        raise ApiError(404, f"Station {station_id} not found")
    return respond({"message": "Station updated",
                    "url": request.host_url + "api/v1.0/stations/" + station_id}, 200)

    if result.deleted_count == 0:
        raise ApiError(404, f"Station {station_id} not found")

def haversine(lat1, lng1, lat2, lng2):
    r = 6371
    p1, p2 = math.radians(lat1), math.radians(lat2)
    dp, dl = math.radians(lat2 - lat1), math.radians(lng2 - lng1)
    a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2
    return 2 * r * math.asin(math.sqrt(a))

    "brakes": ["Front brake feels spongy", "Rear brake not gripping", "Brake lever loose", "Brakes squeal loudly"],
    "tyre": ["Front tyre flat", "Rear tyre puncture", "Tyre worn down to the tread", "Slow leak in rear tyre"],
    "chain": ["Chain keeps slipping", "Chain came off", "Chain rusty and noisy", "Gears skipping"],
    "battery": ["Battery drains very quickly", "Battery won't charge", "Motor cuts out on hills", "Display shows battery error"],
    "other": ["Saddle won't adjust", "Bell missing", "Front light not working", "Basket bent", "Dock lock stuck"],
}
MECH_NOTES = ["Inspected bike at station", "Ordered replacement part", "Replaced part and tested",
              "Adjusted and road tested", "Cleaned and lubricated", "Safety check passed"]
reports = []
def add_report(bike, status, created):
    itype = "battery" if bike["type"] == "electric" and random.random() < 0.4 else \
            random.choice(["brakes", "tyre", "chain", "other"])
    reporter = random.choice(riders + mechanics[:3])
    mech = random.choice(mechanics)
    notes, resolved = [], None
    if status in ("in_progress", "resolved"):
        t = created + dt.timedelta(hours=random.randint(1, 20))
        notes.append({"_id": oid(), "mechanic_id": mech["user_id"], "note": MECH_NOTES[0], "created_at": t})
        if random.random() < 0.5:
            t += dt.timedelta(hours=random.randint(1, 24))
            notes.append({"_id": oid(), "mechanic_id": mech["user_id"], "note": MECH_NOTES[1], "created_at": t})
        if status == "resolved":
            resolved = t + dt.timedelta(hours=random.randint(1, 30))
            notes.append({"_id": oid(), "mechanic_id": mech["user_id"], "note": random.choice(MECH_NOTES[2:]), "created_at": resolved})
    reports.append({
        "report_id": f"MR{len(reports)+1:04d}", "bike_id": bike["bike_id"],
        "reported_by": reporter["user_id"], "issue_type": itype,
        "description": random.choice(ISSUES[itype]),
        "status": status,
        "assigned_to": mech["user_id"] if status != "open" else None,
        "created_at": created, "resolved_at": resolved,
        "notes": notes,   # embedded sub-documents
    })

for b in [b for b in bikes if b["status"] == "under_repair"]:     # live faults
    add_report(b, random.choice(["open", "open", "in_progress"]),
               NOW - dt.timedelta(hours=random.randint(2, 96)))
repeat_offenders = random.sample([b for b in bikes if b["status"] == "available"], 12)
for b in repeat_offenders:                                         # repeated faults
    for _ in range(random.randint(3, 5)):
        add_report(b, "resolved", NOW - dt.timedelta(days=random.randint(5, 85)))
while len(reports) < 220:                                          # one-off historic faults
    b = random.choice([b for b in bikes if b["status"] != "under_repair"])
    add_report(b, "resolved", NOW - dt.timedelta(days=random.randint(5, 85)))
reports.sort(key=lambda r: r["created_at"])
for i, r in enumerate(reports):
    r["report_id"] = f"MR{i+1:04d}"

# ---------------------------------------------------------------- write
def to_json(o):
    if isinstance(o, dict): return {k: to_json(v) for k, v in o.items()}
    if isinstance(o, list): return [to_json(v) for v in o]
    if isinstance(o, dt.datetime): return iso(o)
    return o
